cmake_minimum_required(VERSION 3.24)
project(ImageClassifier VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets Sql Concurrent Multimedia Test)
qt_standard_project_setup()
add_library(classifier_core STATIC src/Store.cpp src/FileOps.cpp src/Media.cpp src/CropGeometry.cpp src/Backend.cpp)
target_include_directories(classifier_core PUBLIC src)
target_link_libraries(classifier_core PUBLIC Qt6::Core Qt6::Gui Qt6::Sql Qt6::Concurrent)
target_compile_definitions(classifier_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(ImageClassifier WIN32 src/main.cpp src/ImageProvider.cpp)
target_link_libraries(ImageClassifier PRIVATE classifier_core Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Multimedia)
target_compile_definitions(ImageClassifier PRIVATE APP_UI_DIR="${CMAKE_CURRENT_SOURCE_DIR}/ui")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE classifier_core Qt6::Test)
add_test(NAME core_tests COMMAND core_tests)
set_tests_properties(core_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(ui_tests tests/ui_tests.cpp src/ImageProvider.cpp)
target_link_libraries(ui_tests PRIVATE classifier_core Qt6::Quick Qt6::QuickControls2 Qt6::Test)
target_compile_definitions(ui_tests PRIVATE TEST_UI_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
add_test(NAME ui_tests COMMAND ui_tests)
set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
