cmake_minimum_required(VERSION 3.24)
project(ImageClassifier VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets Sql Concurrent Multimedia)
qt_standard_project_setup()
add_library(classifier_core STATIC src/Store.cpp src/FileOps.cpp src/Media.cpp src/MediaTime.cpp src/ThumbnailCache.cpp src/LivePhoto.cpp src/AppleLivePhoto.cpp src/HeifImage.cpp src/CropGeometry.cpp src/ModelInstaller.cpp src/SmartCrop.cpp src/Backend.cpp src/AppRestart.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.tools/media/heif/include/libheif/heif.h")
    target_include_directories(classifier_core PRIVATE .tools/media/heif/include)
    target_compile_definitions(classifier_core PRIVATE IMAGECLASSIFIER_HEIF)
endif()
target_include_directories(classifier_core PUBLIC src)
target_link_libraries(classifier_core PUBLIC Qt6::Core Qt6::Gui Qt6::Sql Qt6::Concurrent ole32 shell32 uuid)
target_compile_definitions(classifier_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_definitions(classifier_core PRIVATE APP_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(ImageClassifier WIN32 src/main.cpp src/ImageProvider.cpp)
if(WIN32)
    enable_language(RC)
    configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(ImageClassifier PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
qt_add_resources(ImageClassifier app_brand
    PREFIX "/brand"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/ui/brand"
    FILES ui/brand/guitu.ico
)
target_link_libraries(ImageClassifier PRIVATE classifier_core Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Multimedia)
target_compile_definitions(ImageClassifier PRIVATE APP_UI_DIR="${CMAKE_CURRENT_SOURCE_DIR}/ui")
option(IMAGECLASSIFIER_BUILD_TESTS "Build local-only tests" OFF)
if(IMAGECLASSIFIER_BUILD_TESTS)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
        message(FATAL_ERROR "Local tests are not present. Build without IMAGECLASSIFIER_BUILD_TESTS.")
    endif()
    find_package(Qt6 6.10 REQUIRED COMPONENTS Test)
    enable_testing()
    add_subdirectory(tests)
endif()
